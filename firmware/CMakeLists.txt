cmake_minimum_required(VERSION 3.20)
project(vault_firmware_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(VAULT_SANITIZE "Enable native address and undefined behavior sanitizers" OFF)
if(VAULT_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_compile_options(-Wall -Wextra -Wpedantic)
add_subdirectory(components/vault_json)
add_subdirectory(components/vault_protocol)
add_subdirectory(components/vault_display)
add_subdirectory(components/vault_device)
add_subdirectory(components/vault_actuator)
add_subdirectory(components/vault_transport)
enable_testing()
add_subdirectory(tests)
